<script>
(function () {
  var buttons = Array.prototype.slice.call(document.querySelectorAll('#publicationTagFilter .tag-btn'));
  var groups = Array.prototype.slice.call(document.querySelectorAll('.publication-group'));
  var items = Array.prototype.slice.call(document.querySelectorAll('.publication-entry'));
  var emptyState = document.querySelector('.publication-empty');

  if (!buttons.length || !items.length) {
    return;
  }

  var rules = [
    { tag: 'generation', regex: /(large language model|llm|generative|generation|generate|deliberation|debate|reasoning|knowledge graph|coreference|evolve)/i },
    { tag: 'multimodal', regex: /(multimodal|visual question answering|audio-visual|visual|image|video|vqa|diagram|textbook question answering)/i },
    { tag: 'medical', regex: /(medical|biomedical|health|clinical|medicine|medical imaging)/i },
    { tag: 'world-model', regex: /(world model|navigation|goal navigation|embodied|robot|planning)/i },
    { tag: 'benchmark', regex: /(benchmark|dataset|survey|challenge|evaluation)/i }
  ];

  function normalizeText(value) {
    return (value || '')
      .toString()
      .toLowerCase()
      .replace(/[^\w\u4e00-\u9fa5]+/g, ' ')
      .trim();
  }

  function inferTags(item) {
    var titleNode = item.querySelector('.show_paper_citations');
    var anchor = item.querySelector('a');
    var title = '';

    if (titleNode && titleNode.getAttribute('data-title')) {
      title = titleNode.getAttribute('data-title');
    } else if (anchor) {
      title = anchor.textContent || '';
    } else {
      title = item.textContent || '';
    }

    var combinedText = normalizeText(title + ' ' + item.textContent);
    var tags = [];

    rules.forEach(function (rule) {
      if (rule.regex.test(combinedText)) {
        tags.push(rule.tag);
      }
    });

    item.dataset.tags = tags.join(' ');
    item.dataset.title = title;
  }

  function updateCounts() {
    buttons.forEach(function (button) {
      var tag = button.getAttribute('data-tag');
      var countElement = button.querySelector('.tag-count');
      if (!countElement) {
        return;
      }
      if (tag === 'all') {
        countElement.textContent = items.length;
        return;
      }
      var count = items.filter(function (item) {
        return (item.dataset.tags || '').split(/\s+/).indexOf(tag) !== -1;
      }).length;
      countElement.textContent = count;
    });
  }

  function updateGroups() {
    groups.forEach(function (group) {
      var visibleItems = group.querySelectorAll('.publication-entry:not(.is-filtered-out)');
      group.classList.toggle('is-filtered-out', visibleItems.length === 0);
    });
  }

  function applyFilter(tag) {
    items.forEach(function (item) {
      var tags = (item.dataset.tags || '').split(/\s+/).filter(Boolean);
      var match = tag === 'all' || tags.indexOf(tag) !== -1;
      item.classList.toggle('is-filtered-out', !match);
    });

    buttons.forEach(function (button) {
      button.classList.toggle('active', button.getAttribute('data-tag') === tag);
    });

    updateGroups();

    if (emptyState) {
      var visibleCount = items.filter(function (item) {
        return !item.classList.contains('is-filtered-out');
      }).length;
      emptyState.hidden = visibleCount !== 0;
    }
  }

  items.forEach(inferTags);
  updateCounts();
  applyFilter('all');

  buttons.forEach(function (button) {
    button.addEventListener('click', function () {
      applyFilter(button.getAttribute('data-tag'));
    });
  });
})();
</script>
